cmake_minimum_required(VERSION 3.19)
project(collabtext-consumer-smoke LANGUAGES CXX)

# An independent consumer of the *installed* collabtext alpha.
#
# Roadmap task A06a. Two rules make this a test rather than a demonstration:
#
#   1. Nothing here refers to the collabtext source tree. No
#      target_include_directories pointing at ../libs, no CMAKE_PREFIX_PATH
#      pointing at a build directory. The only thing this project knows about is
#      the install prefix it is given.
#   2. It is registered with CTest, so `ctest --test-dir` has something to run and
#      a zero-match selection is an error rather than a quiet success.

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(CONSUMER_WANT_TRANSPORT "Also require the file-transport component" OFF)
option(CONSUMER_WANT_QT "Also require the Qt components" OFF)
option(CONSUMER_WANT_MISSING "Deliberately request a component the producer did not build" OFF)

set(_wanted core)
if(CONSUMER_WANT_TRANSPORT)
    list(APPEND _wanted file-transport)
endif()
if(CONSUMER_WANT_QT)
    list(APPEND _wanted qt)
endif()
if(CONSUMER_WANT_MISSING)
    # A negative check: asking for a component that does not exist must fail
    # loudly rather than configure to something that cannot work.
    list(APPEND _wanted no-such-component)
endif()

find_package(collabtext CONFIG REQUIRED COMPONENTS ${_wanted})
message(STATUS "consumer: collabtext ${collabtext_VERSION}, enabled: ${collabtext_enabled_components}")

enable_testing()

add_executable(consumer_main main.cpp)
target_link_libraries(consumer_main PRIVATE CollabText::Core)

# Every installed public header, compiled as the first and only library header
# in its own translation unit, from the install tree. This is the check that a
# header really is self-sufficient: the consumer has no way to add a missing
# include path even if it wanted to.
file(GLOB COLLABTEXT_INSTALLED_HEADERS
     "${collabtext_DIR}/../../../include/collabtext/*.h")
if(NOT COLLABTEXT_INSTALLED_HEADERS)
    message(FATAL_ERROR "consumer: found no installed headers next to ${collabtext_DIR}")
endif()

# The three headers that need Qt. Kept as a named list rather than "compile
# everything and hope", because a consumer that asked for Core has no Qt to give
# them, and silently skipping whatever fails to configure is how a packaging
# regression turns into a green run. `tools/check-public-headers.sh` prints this
# same classification from the real transitive includes, so a header that starts
# needing Qt shows up as a diff there.
set(CONSUMER_QT_ONLY_HEADERS CollabDocument.h PresenceManager.h SyncManager.h)

set(_headers_to_check)
foreach(_header ${COLLABTEXT_INSTALLED_HEADERS})
    get_filename_component(_file ${_header} NAME)
    if(_file IN_LIST CONSUMER_QT_ONLY_HEADERS AND NOT CONSUMER_WANT_QT)
        continue()
    endif()
    list(APPEND _headers_to_check "${_file}")
endforeach()

set(_header_tus)
foreach(_file ${_headers_to_check})
    get_filename_component(_stem ${_file} NAME_WE)
    string(MAKE_C_IDENTIFIER "${_stem}" _ident)
    set(_tu "${CMAKE_CURRENT_BINARY_DIR}/only_${_stem}.cpp")
    file(WRITE "${_tu}"
         "// Generated by the consumer smoke: one public header, alone, from the\n"
         "// installed tree. If this needs an include path the consumer does not\n"
         "// have, the header is not self-sufficient.\n"
         "#include <collabtext/${_file}>\n"
         "int consumer_header_${_ident}() { return 0; }\n")
    list(APPEND _header_tus "${_tu}")
endforeach()
add_library(consumer_installed_headers OBJECT ${_header_tus})
target_link_libraries(consumer_installed_headers PRIVATE CollabText::Core)

# Deliberately no CTest entry for consumer_installed_headers. Its evidence is
# that it compiled: a `cmake -E true` placeholder would have reported success
# whether or not a single header was self-sufficient, which is exactly the kind of
# green that means nothing. The verification script fails on a build error, and
# this project configuring at all means the header glob found real files.
list(LENGTH _header_tus _header_count)
list(LENGTH COLLABTEXT_INSTALLED_HEADERS _installed_count)
math(EXPR _skipped "${_installed_count} - ${_header_count}")
if(_skipped GREATER 3)
    message(FATAL_ERROR
        "consumer: only ${_header_count} of ${_installed_count} installed headers "
        "would be compiled. A short glob means the install tree is incomplete, not "
        "that the surface shrank.")
endif()
message(STATUS
    "consumer: compiling ${_header_count} of ${_installed_count} installed public "
    "headers, one TU each (Qt-only headers skipped without -DCONSUMER_WANT_QT=ON)")

add_test(NAME consumer_main COMMAND consumer_main)

if(CONSUMER_WANT_TRANSPORT)
    target_link_libraries(consumer_main PRIVATE CollabText::FileTransport)
    target_compile_definitions(consumer_main PRIVATE CONSUMER_WANT_TRANSPORT=1)
endif()
if(CONSUMER_WANT_QT)
    target_link_libraries(consumer_main PRIVATE CollabText::Widgets Qt6::Widgets)
    target_link_libraries(consumer_installed_headers PRIVATE Qt6::Widgets)
endif()

# The worker consumes only the relocated installation. JSON fixtures are data,
# not source include paths or an engine substitute.
set(CONSUMER_PROBE_GATE "package" CACHE STRING "package/baseline/live")
if(CONSUMER_PROBE_GATE STREQUAL "baseline" OR CONSUMER_PROBE_GATE STREQUAL "live")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(consumer_wave_probes wave-contract/probes.cpp)
    target_link_libraries(consumer_wave_probes PRIVATE CollabText::Core)
    add_test(NAME consumer_wave_contract COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/wave-contract/run.py
        --worker $<TARGET_FILE:consumer_wave_probes>
        --cases ${CMAKE_CURRENT_SOURCE_DIR}/wave-contract/cases.json
        --fixtures ${CONSUMER_PROBE_FIXTURES}
        --manifest ${CONSUMER_CAPABILITY_MANIFEST}
        --revision ${CONSUMER_PRODUCER_REVISION}
        --mode ${CONSUMER_BUILD_MODE}
        --gate ${CONSUMER_PROBE_GATE}
        --output ${CMAKE_CURRENT_BINARY_DIR}/wave-probe-results)
    set_tests_properties(consumer_wave_contract PROPERTIES TIMEOUT 120)
endif()

# W03's second consumer, live gate only: it is a different composition reaching
# the same installed library -- Markdown bytes, a sibling block map and an
# optional transport -- and it shares no adapter, support header or expectation
# table with the Wave-shaped fixture above. It has no allowlist: every case is a
# strict pass or the gate fails. Leaving it out of the baseline gate keeps W02's
# recorded evidence exactly as W02 defined it.
if(CONSUMER_PROBE_GATE STREQUAL "live")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(consumer_markdown_probes markdown-contract/markdown.cpp)
    target_link_libraries(consumer_markdown_probes PRIVATE CollabText::Core)
    add_test(NAME consumer_markdown_contract COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/markdown-contract/run.py
        --worker $<TARGET_FILE:consumer_markdown_probes>
        --cases ${CMAKE_CURRENT_SOURCE_DIR}/markdown-contract/cases.json
        --revision ${CONSUMER_PRODUCER_REVISION}
        --mode ${CONSUMER_BUILD_MODE}
        --output ${CMAKE_CURRENT_BINARY_DIR}/markdown-probe-results)
    set_tests_properties(consumer_markdown_contract PROPERTIES TIMEOUT 120)
endif()

add_executable(consumer_wave_core wave_core.cpp)
target_link_libraries(consumer_wave_core PRIVATE CollabText::Core)
add_test(NAME consumer_wave_core COMMAND consumer_wave_core)

add_executable(consumer_markoff_core markoff_core.cpp)
target_link_libraries(consumer_markoff_core PRIVATE CollabText::Core)
add_test(NAME consumer_markoff_core COMMAND consumer_markoff_core)
